\banswer{
\begin{enumerate}
\item 先计算对易子：
\[
 [A,B]=i(c-2)\begin{pmatrix}0&1\\1&0\end{pmatrix}.
\]
若 $c\ne2$，$A$ 的两个本征值不同，其本征态只能为 $(1,0)^T$、$(0,1)^T$，
而 $B$ 把这两个态互相变换，因此没有共同本征态。$c=2$ 时 $A=I$，任意态都是
$A$ 的本征态，可以选 $B$ 的本征基。所以 $\boxed{c_0=2}$。
\item 由 $\det(B-bI)=b^2-1=0$，得 $b=\pm1$。设本征矢为 $(u,v)^T$，
第一行 $-iv=bu$ 给出 $v=ibu$，归一化后
\[
 \boxed{|b_+\rangle=\frac1{\sqrt2}\begin{pmatrix}1\\i\end{pmatrix},\qquad
        |b_-\rangle=\frac1{\sqrt2}\begin{pmatrix}1\\-i\end{pmatrix}}.
\]
两态的 $A$ 本征值均为 $1$。
\item 当 $c\ne2$，按 $A$ 本征值 $1,c-1$ 排列旧基，按 $B$ 本征值 $+1,-1$ 排列新基。
若旧坐标为 $\psi_A$，新坐标由 $\psi_B=(\langle b_+|\psi\rangle,
\langle b_-|\psi\rangle)^T$ 给出，故
\[
 \boxed{\psi_B=S\psi_A,\quad S=\frac1{\sqrt2}
 \begin{pmatrix}1&-i\\1&i\end{pmatrix}},\qquad SS^\dagger=I.
\]
反向变换为 $\psi_A=S^\dagger\psi_B$；算符变换为 $O_B=SO_AS^\dagger$。
基矢重新排序或乘相位会改变矩阵外观，使用时须固定同一约定。
\end{enumerate}

\item 	
假设粒子间的相互作用 $\mathrm{V}=\lambda \delta\left(x_{1}-x_{2}\right)$ 的微弱相互作用, 求系统的三个最低能量（$ \lambda $的一级近似）及相应的归一化波函数。
\end{enumerate}

\banswer{
\begin{enumerate}
	%\renewcommand{\labelenumi}{\arabic{enumi}.}
	% A(\Alph) a(\alph) I(\Roman) i(\roman) 1(\arabic)
	%设定全局标号series=example	%引用全局变量resume=example
	%[topsep=-0.3em,parsep=-0.3em,itemsep=-0.3em,partopsep=-0.3em]
	%可使用leftmargin调整列表环境左边的空白长度 [leftmargin=0em]
	\item
	记
	\[
		\phi_n(x)=\sqrt{\frac2L}\sin\frac{n\pi x}{L},\qquad
		\epsilon=\frac{\pi^2\hbar^2}{2\mu L^2}.
	\]
	无扰直积态
	$\psi_{n_1n_2}(x_1,x_2)=\phi_{n_1}(x_1)\phi_{n_2}(x_2)$，
	能量为 $(n_1^2+n_2^2)\epsilon$。三个最低能量及态为
	\[
		\boxed{2\epsilon:\ |11\rangle},\qquad
		\boxed{5\epsilon:\ |12\rangle,\ |21\rangle},\qquad
		\boxed{8\epsilon:\ |22\rangle}.
	\]
	\item 
	基态和第二激发态非简并，并利用
	$\int_0^L|\phi_n|^4dx=\frac4{L^2}\int_0^L\sin^4(n\pi x/L)dx=3/(2L)$，得
	\[
		\boxed{E_{\rm g}=2\epsilon+\frac{3\lambda}{2L},
		\qquad \Psi_{\rm g}^{(0)}=|11\rangle},
	\]
	\[
		\boxed{E_{\rm 2nd}=8\epsilon+\frac{3\lambda}{2L},
		\qquad \Psi_{\rm 2nd}^{(0)}=|22\rangle}.
	\]
	在 $|12\rangle,|21\rangle$ 子空间，由 $\delta(x_1-x_2)$ 先积掉一个坐标，
	\[
	 \langle12|V|12\rangle=\langle12|V|21\rangle
	 =\lambda\int_0^L\phi_1^2(x)\phi_2^2(x)dx=\frac\lambda L.
	\]
	其余两元相同，故微扰矩阵为
	\[
		W=\frac{\lambda}{L}\begin{pmatrix}1&1\\1&1\end{pmatrix}.
	\]
	所以第一激发二重态分裂为
	\[
		\boxed{E_-=5\epsilon,\qquad
		\Psi_-^{(0)}=\frac{|12\rangle-|21\rangle}{\sqrt2}},
	\]
	\[
		\boxed{E_+=5\epsilon+\frac{2\lambda}{L},\qquad
		\Psi_+^{(0)}=\frac{|12\rangle+|21\rangle}{\sqrt2}}.
	\]
	这里列出的波函数是决定一级能量所需的零级归一化态。若还要求波函数本身准确到
	$O(\lambda)$，对每个已经对角化的态 $|\alpha\rangle$ 再加
	\[
		|\alpha^{(1)}\rangle
		=\sum_{\beta\notin{\cal D}_\alpha}
		\frac{\langle\beta|V|\alpha\rangle}
		{E_\alpha^{(0)}-E_\beta^{(0)}}|\beta\rangle
	\]
	即可。这里 $\beta=|m,n\rangle$ 遍历未扰直积态，排除同一无扰简并子空间。
	为使该通式可直接计算，记
	\[
	 I^{(r)}_{mn}=\int_0^L\phi_m\phi_n\phi_r^2dx
	 =\frac1L\left[\delta_{mn}-\frac12\delta_{|m-n|,2r}
	                              +\frac12\delta_{m+n,2r}\right].
	\]
	此式由 $2\sin A\sin B=\cos(A-B)-\cos(A+B)$ 的正交积分得到。
	对 $r=1,2$，完整的一级波函数可写为
	\[
	 |\Psi_r\rangle=|rr\rangle+
	 \sum_{(m,n)\ne(r,r)}\frac{\lambda I^{(r)}_{mn}}
	 {\epsilon(2r^2-m^2-n^2)}|mn\rangle+O(\lambda^2).
	\]
	对 $|+\rangle=(|12\rangle+|21\rangle)/\sqrt2$，相应的矩阵元为
	\[
	 J_{mn}=\langle mn|V|+\rangle
	 =\frac{\lambda}{\sqrt2L}
	 [\delta_{|m-n|,1}-\delta_{|m-n|,3}+\delta_{m+n,3}],
	\]
	因此
	\[
	 |\Psi_+\rangle=|+\rangle+
	 \sum_{(m,n)\notin\{(1,2),(2,1)\}}
	 \frac{J_{mn}}{\epsilon(5-m^2-n^2)}|mn\rangle+O(\lambda^2).
	\]
	反对称态 $|-\rangle$ 在 $x_1=x_2$ 上为零，故 $V|-\rangle=0$，是精确不变态。
	这些修正均与原零级态正交，范数为 $1+O(\lambda^2)$，因此在一级精度下已归一化。
\end{enumerate}

\item 
一个质量为 $\mu$ 、电荷为 $q$ 的谐振子在外电场 $\varepsilon$ 的作用下 $ \hat{ \mathrm{H}}=\frac{p^{2}}{2 \mu}+\frac{1}{2} \mu \omega^{2} x^{2}-q \varepsilon x $。
\begin{enumerate}
	%\renewcommand{\labelenumi}{\arabic{enumi}.}
	% A(\Alph) a(\alph) I(\Roman) i(\roman) 1(\arabic)
	%设定全局标号series=example	%引用全局变量resume=example
	%[topsep=-0.3em,parsep=-0.3em,itemsep=-0.3em,partopsep=-0.3em]
	%可使用leftmargin调整列表环境左边的空白长度 [leftmargin=0em]
	\item
	若已知 $\hat{H_{0}}=\frac{P^{2}}{2 \mu}+\frac{1}{2} \mu \omega^{2} x^{2}$ 的本征能量为 $\mathrm{E}_{\mathrm{n}}$ 的相应本征函数为 $ \ket{n} $, 求 $\hat{\mathrm{H}}$ 的  本征
	能量 $\widetilde{\mathrm{E}}_{\mathrm{n}}$及相应的本征函数 $ \ket{\tilde{n}} $。
	\item 
	若 $\mathrm{t}=0$ 时刻处于系统 $\hat{H_{0}}$ 的初态 $ \ket{0} $, 求 $\mathrm{t}>0$ 时仍处 $\hat{H_{0}}$ 的初态 $ \ket{0} $ 几率 $P_{0}$。
	
\end{enumerate}

\banswer{
\begin{enumerate}
	\item 配方求能谱和本征函数。记
	\[
	 x_0=\frac{q\varepsilon}{\mu\omega^2},\qquad
	 \Delta=\frac{q^2\varepsilon^2}{2\mu\omega^2}
	       =\frac12\mu\omega^2x_0^2,
	\]
	则
	\[
	 H=\frac{p^2}{2\mu}+\frac12\mu\omega^2(x-x_0)^2-\Delta.
	\]
	因此势阱中心由 $0$ 移到 $x_0$，频率不变，所有能级共同下移 $\Delta$：
	\[
	 \boxed{\widetilde E_n=E_n-\Delta
	        =\hbar\omega\left(n+\frac12\right)-\Delta},
	 \qquad n=0,1,2,\ldots.
	\]
	令 $T(x_0)=e^{-ix_0p/\hbar}$。在位置表象中，$p=-i\hbar\partial_x$，故
	\[
	 \langle x|T(x_0)|n\rangle
	 =e^{-x_0\partial_x}\varphi_n(x)=\varphi_n(x-x_0).
	\]
	平移保持归一化，且将原本征方程中的 $x$ 换成 $x-x_0$ 后即可验证
	\[
	 \boxed{|\widetilde n\rangle=T(x_0)|n\rangle},\qquad
	 \boxed{\widetilde\varphi_n(x)=\varphi_n(x-x_0)}.
	\]

	\item 题目要求仍处于\textbf{原 Hamilton 量 $H_0$ 的基态 $|0\rangle$} 的概率，
	应将初态展开到新 Hamilton 量 $H$ 的能量本征基中，再进行时间演化。

	\textbf{第一步：建立新、旧升降算符的关系。}
	定义
	\[
	 a=\frac{\mu\omega x+ip}{\sqrt{2\mu\hbar\omega}},\qquad
	 \widetilde a=\frac{\mu\omega(x-x_0)+ip}{\sqrt{2\mu\hbar\omega}},
	 \qquad \gamma=\sqrt{\frac{\mu\omega}{2\hbar}}x_0.
	\]
	其中 $\gamma$ 为实数，可正可负，并满足 $\Delta=\hbar\omega\gamma^2$。于是
	\[
	 \widetilde a=a-\gamma,\qquad
	 \widetilde a^\dagger=a^\dagger-\gamma,\qquad
	 |\widetilde n\rangle=
	 \frac{(\widetilde a^\dagger)^n}{\sqrt{n!}}|\widetilde0\rangle.
	\]

	\textbf{第二步：计算初态的展开系数。}
	写成
	\[
	 |0\rangle=\sum_{n=0}^\infty c_n|\widetilde n\rangle,\qquad
	 c_n=\langle\widetilde n|0\rangle.
	\]
	利用 $a|0\rangle=0$，有 $(a-\gamma)^n|0\rangle=(-\gamma)^n|0\rangle$，故
	\[
	 c_n=\frac{\langle\widetilde0|\widetilde a^{\,n}|0\rangle}{\sqrt{n!}}
	     =\frac{(-\gamma)^n}{\sqrt{n!}}c_0.
	\]
	剩下的 $c_0$ 直接由两个基态的 Gaussian 重叠积分求出。
	令 $b=\mu\omega/\hbar$，取实的归一化基态函数
	\[
	 \varphi_0(x)=\left(\frac b\pi\right)^{1/4}e^{-bx^2/2},\qquad
	 \widetilde\varphi_0(x)=\left(\frac b\pi\right)^{1/4}e^{-b(x-x_0)^2/2}.
	\]
	由于 $(x-x_0)^2+x^2=2(x-x_0/2)^2+x_0^2/2$，
	\begin{align*}
	 c_0&=\sqrt{\frac b\pi}\int_{-\infty}^\infty
	       e^{-b[(x-x_0)^2+x^2]/2}\,dx\\
	    &=e^{-bx_0^2/4}\sqrt{\frac b\pi}\int_{-\infty}^\infty
	       e^{-b(x-x_0/2)^2}\,dx
	     =e^{-bx_0^2/4}=e^{-\gamma^2/2}.
	\end{align*}
	所以
	\[
	 \boxed{c_n=e^{-\gamma^2/2}\frac{(-\gamma)^n}{\sqrt{n!}}},\qquad
	 |c_n|^2=e^{-\gamma^2}\frac{\gamma^{2n}}{n!}.
	\]
	可核对 $\sum_{n=0}^\infty|c_n|^2=e^{-\gamma^2}e^{\gamma^2}=1$。

	\textbf{第三步：演化后投影回原基态。}
	在 $t>0$ 时，外电场保持不变，故各新能量本征态独立积累相位：
	\[
	 |\psi(t)\rangle=\sum_{n=0}^\infty
	 c_n e^{-i\widetilde E_nt/\hbar}|\widetilde n\rangle.
	\]
	因为 $\langle0|\widetilde n\rangle=c_n^*$，返回原基态的振幅为
	\begin{align*}
	 A_0(t)&=\langle0|\psi(t)\rangle
	 =\sum_{n=0}^\infty|c_n|^2e^{-i\widetilde E_nt/\hbar}\\
	 &=e^{-i(\omega/2-\Delta/\hbar)t}e^{-\gamma^2}
	   \sum_{n=0}^\infty\frac{(\gamma^2e^{-i\omega t})^n}{n!}\\
	 &=e^{-i(\omega/2-\Delta/\hbar)t}
	   \exp\!\left[\gamma^2(e^{-i\omega t}-1)\right].
	\end{align*}
	求和使用了指数函数的幂级数。取模平方时，共同相位消失，
	余下复指数只需取指数的两倍实部，故
	\begin{align*}
	 P_0(t)&=|A_0(t)|^2
	 =\exp\!\left[2\gamma^2(\cos\omega t-1)\right]\\
	 &=\boxed{\exp\!\left[-\frac{2\mu\omega x_0^2}{\hbar}
	                 \sin^2\frac{\omega t}{2}\right]}\\
	 &=\boxed{\exp\!\left[-\frac{2q^2\varepsilon^2}{\mu\hbar\omega^3}
	                 \sin^2\frac{\omega t}{2}\right]}.
	\end{align*}
	结果为精确解：$P_0(0)=1$，无电场时恒为 $1$；每隔 $2\pi/\omega$ 返回原基态，
	而 $t=(2k+1)\pi/\omega$ 时概率最小。
\end{enumerate}
}
