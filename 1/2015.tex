\item 
证明 $\left[H, \hat{\boldsymbol{S}}^{\mathbf{2}}\right]=0$

\banswer{
\begin{enumerate}
	%\renewcommand{\labelenumi}{\arabic{enumi}.}
	% A(\Alph) a(\alph) I(\Roman) i(\roman) 1(\arabic)
	%设定全局标号series=example	%引用全局变量resume=example

	%可使用leftmargin调整列表环境左边的空白长度 [leftmargin=0em]
	\item
		由 $\hat{\boldsymbol S}=\hat{\boldsymbol S}_1+\hat{\boldsymbol S}_2$ 以及
		$\hat{\boldsymbol S}_1^2=\hat{\boldsymbol S}_2^2=\frac{3}{4}\hbar^2$，有
		\[
		\hat{\boldsymbol S}^{\,2}
		=\hat{\boldsymbol S}_1^2+\hat{\boldsymbol S}_2^2
		+2\hat{\boldsymbol S}_1\cdot\hat{\boldsymbol S}_2,
		\]
		故
		\[
		\hat{\boldsymbol S}_1\cdot\hat{\boldsymbol S}_2
		=\frac{1}{2}\hat{\boldsymbol S}^{\,2}-\frac{3}{4}\hbar^2.
		\]
		另一方面，
		\[
		(\hat{\boldsymbol S}\cdot\hat{\boldsymbol n})^2
		=(\hat{\boldsymbol S}_1\cdot\hat{\boldsymbol n})^2
		+(\hat{\boldsymbol S}_2\cdot\hat{\boldsymbol n})^2
		+2(\hat{\boldsymbol S}_1\cdot\hat{\boldsymbol n})
		 (\hat{\boldsymbol S}_2\cdot\hat{\boldsymbol n}).
		\]
		对自旋 $1/2$ 粒子，
		$\hat{\boldsymbol S}_i\cdot\hat{\boldsymbol n}
		=\frac{\hbar}{2}\boldsymbol\sigma_i\cdot\hat{\boldsymbol n}$，且
		$(\boldsymbol\sigma_i\cdot\hat{\boldsymbol n})^2=1$，所以
		$(\hat{\boldsymbol S}_i\cdot\hat{\boldsymbol n})^2=\frac{1}{4}\hbar^2$。于是
		\[
		(\hat{\boldsymbol S}_1\cdot\hat{\boldsymbol n})
		(\hat{\boldsymbol S}_2\cdot\hat{\boldsymbol n})
		=\frac{1}{2}(\hat{\boldsymbol S}\cdot\hat{\boldsymbol n})^2-\frac{1}{4}\hbar^2.
		\]

		\item
		将上面两个恒等式代入 $H$，常数项相消，得
		\[
		H=\frac{3}{2}(\hat{\boldsymbol S}\cdot\hat{\boldsymbol n})^2
		-\frac{1}{2}\hat{\boldsymbol S}^{\,2}.
		\]
		$\hat{\boldsymbol n}$ 作用于空间部分，与自旋算符对易；又有
		$[\hat{\boldsymbol S}^{\,2},\hat S_i]=0$。因此
		\[
		[\hat{\boldsymbol S}^{\,2},\hat{\boldsymbol S}\cdot\hat{\boldsymbol n}]=0,
		\qquad [H,\hat{\boldsymbol S}^{\,2}]=0.
		\]
	
\end{enumerate}

\item 
一个二能级体系, 哈密顿算符的矩阵表达式为 $H_{0}=\left(\begin{array}{cc}E_{1}^{(0)} & 0 \\ 0 & E_{2}^{(0)}\end{array}\right),\left(E_{1}^{(0)}<E_{2}^{(0)}\right) $。 设 $t=0$ 时刻体系处
于 $H_{0}$ 的基态上, 后受微扰 $H^{\prime}$ 的作用, $H^{\prime}=\left(\begin{array}{cc}0 & \gamma \\ \gamma & 0\end{array}\right), \gamma$ 为常数。
\begin{enumerate}
	%\renewcommand{\labelenumi}{\arabic{enumi}.}
	% A(\Alph) a(\alph) I(\Roman) i(\roman) 1(\arabic)
	%设定全局标号series=example	%引用全局变量resume=example
	%[topsep=-0.3em,parsep=-0.3em,itemsep=-0.3em,partopsep=-0.3em]
	%可使用leftmargin调整列表环境左边的空白长度 [leftmargin=0em]
	\item
	求出在 $t>0$ 时刻, 体系处于 $H_{0}$ 激发态的几率 $P_{E_{2}^{(0)}}(t)$ 的精确表达式。
	\item 
	利用一阶含时微扰论求 $P_{E_{2}^{(0)}}(t)$, 并与精确表达式比较, 讨论所得结果的适用条件。
	
	
	
\end{enumerate}


\banswer{
\begin{enumerate}
	\item
	记能级间隔 $\Delta=E_2^{(0)}-E_1^{(0)}>0$，并去掉只产生整体相位的
	$\frac{1}{2}(E_1^{(0)}+E_2^{(0)})I$。剩余哈密顿量为
	\[
	\widetilde H=\begin{pmatrix}-\Delta/2&\gamma\\ \gamma&\Delta/2\end{pmatrix},
	\qquad
	\widetilde H^{\,2}=\frac{\Delta^2+4\gamma^2}{4}I.
	\]
	令 $\Omega=\sqrt{\Delta^2+4\gamma^2}/\hbar$，则
	\[
	e^{-i\widetilde Ht/\hbar}
	=I\cos\frac{\Omega t}{2}
	-\frac{2i\widetilde H}{\sqrt{\Delta^2+4\gamma^2}}
	 \sin\frac{\Omega t}{2}.
	\]
	初态为 $|1\rangle=(1,0)^{\mathrm T}$，故跃迁振幅和精确概率分别为
	\[
	\langle2|\psi(t)\rangle
	=-\frac{2i\gamma}{\sqrt{\Delta^2+4\gamma^2}}
	e^{-i(E_1^{(0)}+E_2^{(0)})t/(2\hbar)}
	\sin\frac{\Omega t}{2},
	\]
	\[
	\boxed{P_{E_2^{(0)}}(t)
	=\frac{4\gamma^2}{\Delta^2+4\gamma^2}
	\sin^2\!\left(\frac{\sqrt{\Delta^2+4\gamma^2}}{2\hbar}t\right)}.
	\]

	\item
	一阶含时微扰论给出
	\[
	c_2^{(1)}(t)
	=-\frac{i}{\hbar}\int_0^t\gamma e^{i\Delta t'/\hbar}\,dt'
	=\frac{\gamma}{\Delta}\left(1-e^{i\Delta t/\hbar}\right),
	\]
	从而
	\[
	\boxed{P_{E_2^{(0)}}^{(1)}(t)
	=\frac{4\gamma^2}{\Delta^2}
	\sin^2\!\left(\frac{\Delta t}{2\hbar}\right)}.
	\]
	它正是精确结果在 $|\gamma|/\Delta\ll1$ 时保留到 $\gamma^2$ 的展开；同时应保证所求跃迁概率远小于 $1$，以使一阶近似自洽。
\end{enumerate}
}


\end{enumerate}
