\banswer{
\begin{enumerate}
\item 这是周期边界而非硬墙边界，故不能令端点波函数为零。
对 $E>0$，写 $\psi=Ae^{ikx}+Be^{-ikx}$。两条周期条件相加、相减后分别要求
\[
 A(e^{ikL}-1)=0,\qquad B(e^{-ikL}-1)=0.
\]
非零解要求 $kL=2\pi n$。能量及归一化动量本征态为
\[
 \boxed{E_n=\frac{2\pi^2\hbar^2n^2}{\mu L^2},\qquad
 \phi_n(x)=\frac1{\sqrt L}e^{i2\pi nx/L}},\quad n\in\mathbb Z.
\]
$E=0$ 时通解为 $Ax+B$，周期条件使 $A=0$，留下常数基态；
负能量不可能，因为 $\langle H\rangle=(\hbar^2/2\mu)\int|\psi'|^2dx\ge0$。
因此基态非简并，正能级由 $n$ 与 $-n$ 构成二重简并。
\item 第一激发能级由 $n=\pm1$ 构成，一般归一化态为
\[
 \psi=c_+\phi_++c_-\phi_-,\qquad |c_+|^2+|c_-|^2=1.
\]
因 $p\phi_\pm=\pm(2\pi\hbar/L)\phi_\pm$ 且二态正交，交叉矩阵元为零，故
\[
 \langle p\rangle=\frac{2\pi\hbar}{L}(|c_+|^2-|c_-|^2).
\]
与归一化条件联立，得
$\boxed{|c_+|=|c_-|=1/\sqrt2}$；相对相位任意。
\end{enumerate}

\item 
粒子在球对称谐振子势阱 $V(r)=\frac{1}{2} \mu \omega^{2}\left(x^{2}+y^{2}+z^{2}\right)$ 中运动, 受到微扰作用
\[ 
H^{\prime}=\lambda\left(x y z+x^{2} y+x y^{2}\right)
 \]
其中 $\lambda$ 为常数。求准确到二级微扰修正的基态能量。\\
提示：粒子数表象下, $ \left\langle n_{x}^{\prime}|\hat{x}| n_{x}\right\rangle=\sqrt{\frac{\hbar}{2 \mu \omega}}\left(\sqrt{n_{x}+1} \delta_{n_{x}^{\prime}, n_{x}+1}+\sqrt{n_{x}} \delta_{n_{x}^{\prime}, n_{x}-1}\right) )$。
	
\banswer{
令 $x_0=\sqrt{\hbar/(2\mu\omega)}$，$x=x_0(a_x+a_x^\dagger)$，其他方向类似。
无扰态 $|n_xn_yn_z\rangle$ 的能量为
$E^{(0)}=(n_x+n_y+n_z+3/2)\hbar\omega$，基态为 $|000\rangle$。
微扰各项都是总宇称为奇的三次项，故
\[
 E_0^{(1)}=\langle000|H'|000\rangle=0.
\]
为计算二阶，逐项作用于基态：
\begin{align*}
 xyz|000\rangle&=x_0^3|111\rangle,\\
 x^2y|000\rangle&=x_0^3(|010\rangle+\sqrt2|210\rangle),\\
 xy^2|000\rangle&=x_0^3(|100\rangle+\sqrt2|120\rangle).
\end{align*}
其中用了 $(a+a^\dagger)^2|0\rangle=|0\rangle+\sqrt2|2\rangle$。
五个中间态互相正交；$|010\rangle,|100\rangle$ 的能量分母为 $-\hbar\omega$，
其余三个为 $-3\hbar\omega$。所以
\begin{align*}
 E_0^{(2)}
 &=\lambda^2x_0^6\left[-\frac1{3\hbar\omega}
          -\frac1{\hbar\omega}-\frac2{3\hbar\omega}
          -\frac1{\hbar\omega}-\frac2{3\hbar\omega}\right]\\
 &=-\frac{11\lambda^2x_0^6}{3\hbar\omega}
 =-\frac{11\lambda^2\hbar^2}{24\mu^3\omega^4}.
\end{align*}
因此至二级微扰精度，
\[
 \boxed{E_0\simeq\frac32\hbar\omega-
                  \frac{11\lambda^2\hbar^2}{24\mu^3\omega^4}}.
\]
这给出从无扰基态延拓的微扰能量；三次势在无穷远处不受下界约束，
不能将该形式微扰结果当作全空间势能具有严格基态的证明。
}


\item 
两个自旋为 $\frac{1}{2}$ 的粒子组成的体系, $\hat{\boldsymbol{S}}_{\mathbf{1}}$ 和 $\hat{\boldsymbol{S}}_{\mathbf{2}}$ 分别表示两个粒子的自旋算符, $\hat{\boldsymbol{S}}$ 为两粒子的总自旋算符, $\hat{\boldsymbol{n}}$ 表 示两粒子相对运动方向的单位矢量。设系统的相互作用哈密顿量为 $H=3\left(\hat{\boldsymbol{S}}_{\mathbf{1}} \cdot \hat{\boldsymbol{n}}\right)\left(\hat{\boldsymbol{S}}_{\mathbf{2}} \cdot \hat{\boldsymbol{n}}\right)-\hat{\boldsymbol{S}}_{1} \cdot \hat{\boldsymbol{S}}_{2}$。

\item 
证明 $\left[H, \hat{\boldsymbol{S}}^{\mathbf{2}}\right]=0$	
	
	
\end{enumerate}


\banswer{
\begin{enumerate}
	%\renewcommand{\labelenumi}{\arabic{enumi}.}
	% A(\Alph) a(\alph) I(\Roman) i(\roman) 1(\arabic)
	%设定全局标号series=example	%引用全局变量resume=example

	\item
		由 $\hat{\boldsymbol S}=\hat{\boldsymbol S}_1+\hat{\boldsymbol S}_2$ 以及
		$\hat{\boldsymbol S}_1^2=\hat{\boldsymbol S}_2^2=\frac{3}{4}\hbar^2$，有
		\[
		\hat{\boldsymbol S}^{\,2}
		=\hat{\boldsymbol S}_1^2+\hat{\boldsymbol S}_2^2
		+2\hat{\boldsymbol S}_1\cdot\hat{\boldsymbol S}_2,
		\]
		故
		\[
		\hat{\boldsymbol S}_1\cdot\hat{\boldsymbol S}_2
		=\frac{1}{2}\hat{\boldsymbol S}^{\,2}-\frac{3}{4}\hbar^2.
		\]
		另一方面，
		\[
		(\hat{\boldsymbol S}\cdot\hat{\boldsymbol n})^2
		=(\hat{\boldsymbol S}_1\cdot\hat{\boldsymbol n})^2
		+(\hat{\boldsymbol S}_2\cdot\hat{\boldsymbol n})^2
		+2(\hat{\boldsymbol S}_1\cdot\hat{\boldsymbol n})
		 (\hat{\boldsymbol S}_2\cdot\hat{\boldsymbol n}).
		\]
		对自旋 $1/2$ 粒子，
		$\hat{\boldsymbol S}_i\cdot\hat{\boldsymbol n}
		=\frac{\hbar}{2}\boldsymbol\sigma_i\cdot\hat{\boldsymbol n}$，且
		$(\boldsymbol\sigma_i\cdot\hat{\boldsymbol n})^2=1$，所以
		$(\hat{\boldsymbol S}_i\cdot\hat{\boldsymbol n})^2=\frac{1}{4}\hbar^2$。于是
		\[
		(\hat{\boldsymbol S}_1\cdot\hat{\boldsymbol n})
		(\hat{\boldsymbol S}_2\cdot\hat{\boldsymbol n})
		=\frac{1}{2}(\hat{\boldsymbol S}\cdot\hat{\boldsymbol n})^2-\frac{1}{4}\hbar^2.
		\]

		\item
		将上面两个恒等式代入 $H$，常数项相消，得
		\[
		H=\frac{3}{2}(\hat{\boldsymbol S}\cdot\hat{\boldsymbol n})^2
		-\frac{1}{2}\hat{\boldsymbol S}^{\,2}.
		\]
		$\hat{\boldsymbol n}$ 作用于空间部分，与自旋算符对易；又有
		$[\hat{\boldsymbol S}^{\,2},\hat S_i]=0$。因此
		\[
		[\hat{\boldsymbol S}^{\,2},\hat{\boldsymbol S}\cdot\hat{\boldsymbol n}]=0,
		\qquad [H,\hat{\boldsymbol S}^{\,2}]=0.
		\]
	
\end{enumerate}

\item 
质量为 $\mu$ 的粒子在一维势 $V(x)=\left\{\begin{array}{ll}\infty & x<0 \\ B x & x>0\end{array}\right.$ 中运动, 其中 $B>0$ 是常数。
\begin{enumerate}
	%\renewcommand{\labelenumi}{\arabic{enumi}.}
	% A(\Alph) a(\alph) I(\Roman) i(\roman) 1(\arabic)
	%设定全局标号series=example	%引用全局变量resume=example
	%[topsep=-0.3em,parsep=-0.3em,itemsep=-0.3em,partopsep=-0.3em]
	%可使用leftmargin调整列表环境左边的空白长度 [leftmargin=0em]
	\item
	试从下列波函数中选择一个合理的束缚态试探波函数, 并说明理由。
\threechoices
{$e^{-\frac{x}{a}}$}
{$x e^{-\frac{x}{a}}$}
{$1-e^{-\frac{x}{a}}$}
其中 $a>0$ 为变分参数。

\item 
取所选的试探波函数，用变分法估算体系基态能量。

	
	
	
\end{enumerate}


\banswer{
\begin{enumerate}
\item 无限高墙要求 $\psi(0)=0$，束缚态还须在 $x\to\infty$ 衰减并平方可积。
$e^{-x/a}$ 不满足第一条，$1-e^{-x/a}$ 不满足第二条，故选 $xe^{-x/a}$（$a>0$）。
\item 设 $\psi=Nxe^{-x/a}$（$x\ge0$，负半轴为零）。使用
\[
 \int_0^\infty x^ne^{-2x/a}dx=n!\left(\frac a2\right)^{n+1},
\]
归一化给出 $1=N^2a^3/4$，故 $N=2/a^{3/2}$。
边界项为零，动能可按导数模平方计算：
\begin{align*}
 \langle T\rangle&=\frac{\hbar^2N^2}{2\mu}
        \int_0^\infty(1-x/a)^2e^{-2x/a}dx
        =\frac{\hbar^2}{2\mu a^2},\\
 \langle V\rangle&=BN^2\int_0^\infty x^3e^{-2x/a}dx=\frac{3Ba}{2}.
\end{align*}
故 $E(a)=\hbar^2/(2\mu a^2)+3Ba/2$。
\[
 E'(a)=-\frac{\hbar^2}{\mu a^3}+\frac{3B}{2}=0
 \quad\Longrightarrow\quad
 a_*=\left(\frac{2\hbar^2}{3B\mu}\right)^{1/3}.
\]
$E''(a_*)=3\hbar^2/(\mu a_*^4)>0$，且两端 $a\to0,\infty$ 均使 $E$ 发散，
故该驻点是极小值。由驻点条件 $\langle T\rangle=3Ba_*/4$，
\[
 \boxed{E_{\rm var}=\frac94B\left(\frac{2\hbar^2}{3B\mu}\right)^{1/3}},
 \qquad E_0\le E_{\rm var}.
\]
\end{enumerate}
}



\item 
一个二能级体系, 哈密顿算符的矩阵表达式为 $H_{0}=\left(\begin{array}{cc}E_{1}^{(0)} & 0 \\ 0 & E_{2}^{(0)}\end{array}\right),\left(E_{1}^{(0)}<E_{2}^{(0)}\right) $。 设 $t=0$ 时刻体系处
于 $H_{0}$ 的基态上, 后受微扰 $H^{\prime}$ 的作用, $H^{\prime}=\left(\begin{array}{cc}0 & \gamma \\ \gamma & 0\end{array}\right), \gamma$ 为常数。
\begin{enumerate}
	%\renewcommand{\labelenumi}{\arabic{enumi}.}
	% A(\Alph) a(\alph) I(\Roman) i(\roman) 1(\arabic)
	%设定全局标号series=example	%引用全局变量resume=example
	%[topsep=-0.3em,parsep=-0.3em,itemsep=-0.3em,partopsep=-0.3em]
	%可使用leftmargin调整列表环境左边的空白长度 [leftmargin=0em]
	\item
	求出在 $t>0$ 时刻, 体系处于 $H_{0}$ 激发态的几率 $P_{E_{2}^{(0)}}(t)$ 的精确表达式。
	\item 
	利用一阶含时微扰论求 $P_{E_{2}^{(0)}}(t)$, 并与精确表达式比较, 讨论所得结果的适用条件。
	
	
	
\end{enumerate}


\banswer{
\begin{enumerate}
	\item
	记能级间隔 $\Delta=E_2^{(0)}-E_1^{(0)}>0$，并去掉只产生整体相位的
	$\frac{1}{2}(E_1^{(0)}+E_2^{(0)})I$。剩余哈密顿量为
	\[
	\widetilde H=\begin{pmatrix}-\Delta/2&\gamma\\ \gamma&\Delta/2\end{pmatrix},
	\qquad
	\widetilde H^{\,2}=\frac{\Delta^2+4\gamma^2}{4}I.
	\]
	令 $\Omega=\sqrt{\Delta^2+4\gamma^2}/\hbar$，则
	\[
	e^{-i\widetilde Ht/\hbar}
	=I\cos\frac{\Omega t}{2}
	-\frac{2i\widetilde H}{\sqrt{\Delta^2+4\gamma^2}}
	 \sin\frac{\Omega t}{2}.
	\]
	初态为 $|1\rangle=(1,0)^{\mathrm T}$，故跃迁振幅和精确概率分别为
	\[
	\langle2|\psi(t)\rangle
	=-\frac{2i\gamma}{\sqrt{\Delta^2+4\gamma^2}}
	e^{-i(E_1^{(0)}+E_2^{(0)})t/(2\hbar)}
	\sin\frac{\Omega t}{2},
	\]
	\[
	\boxed{P_{E_2^{(0)}}(t)
	=\frac{4\gamma^2}{\Delta^2+4\gamma^2}
	\sin^2\!\left(\frac{\sqrt{\Delta^2+4\gamma^2}}{2\hbar}t\right)}.
	\]

	\item
	一阶含时微扰论给出
	\[
	c_2^{(1)}(t)
	=-\frac{i}{\hbar}\int_0^t\gamma e^{i\Delta t'/\hbar}\,dt'
	=\frac{\gamma}{\Delta}\left(1-e^{i\Delta t/\hbar}\right),
	\]
	从而
	\[
	\boxed{P_{E_2^{(0)}}^{(1)}(t)
	=\frac{4\gamma^2}{\Delta^2}
	\sin^2\!\left(\frac{\Delta t}{2\hbar}\right)}.
	\]
	对固定的时间，上式正是精确结果在 $|\gamma|/\Delta\ll1$ 时保留到 $\gamma^2$ 的展开。
	但此展开并非对任意长时间一致成立：精确振荡频率比 $\Delta/\hbar$ 多
	$2\gamma^2/(\hbar\Delta)+O(\gamma^4)$。
	若还要准确追踪振荡相位，一个充分条件是 $\gamma^2t/(\hbar\Delta)\ll1$。
	仅有跃迁概率小，并不能消除很长时间后的相位误差。
\end{enumerate}
}


\end{enumerate}
